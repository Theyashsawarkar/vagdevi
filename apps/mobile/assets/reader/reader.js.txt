// upright, the whole page shows; on its side, the page fills the width and scrolls
const fit = (w, h) => {
  const s = W() > H() ? W() / w : Math.min(W() / w, H() / h);
  return [Math.floor(w * s), Math.floor(h * s), s];
};

const cache = new Map();
const CACHE = 7;

async function renderPDF(n) {
  const hit = cache.get(n);
  if (hit) return hit;
  const page = await S.doc.getPage(n);
  const base = page.getViewport({ scale: 1 });
  const [w, h, s] = fit(base.width, base.height);
  const dpr = Math.min(window.devicePixelRatio || 1, 3);
  const vp = page.getViewport({ scale: s * dpr });
  const canvas = document.createElement("canvas");
  canvas.width = vp.width;
  canvas.height = vp.height;
  canvas.style.width = `${w}px`;
  canvas.style.height = `${h}px`;
  await page.render({ canvas, canvasContext: canvas.getContext("2d"), viewport: vp }).promise;
  // the text, invisible over the picture: what a long press reads
  const layer = document.createElement("div");
  layer.className = "textLayer";
  layer.style.setProperty("--total-scale-factor", String(s));
  layer.style.setProperty("--scale-factor", String(s));
  try {
    const tl = new pdfjs.TextLayer({ textContentSource: await page.getTextContent(), container: layer, viewport: page.getViewport({ scale: s }) });
    await tl.render();
  } catch {
    // a scanned page has no text: words just aren't looked up there
  }
  const out = { w, h, canvas, layer };
  cache.set(n, out);
  if (cache.size > CACHE) {
    let far = -1, key = 0;
    for (const k of cache.keys()) {
      const d = Math.abs(k - S.page);
      if (d > far) { far = d; key = k; }
    }
    cache.delete(key);
  }
  return out;
}

const IDB = () =>
  new Promise((ok, no) => {
    const r = indexedDB.open("vagdevi", 1);
    r.onupgradeneeded = () => r.result.createObjectStore("reflow");
    r.onsuccess = () => ok(r.result);
    r.onerror = () => no(r.error);
  });
const idbGet = async (key) => {
  const db = await IDB();
  return new Promise((ok) => {
    const q = db.transaction("reflow").objectStore("reflow").get(key);
    q.onsuccess = () => ok(q.result ?? null);
    q.onerror = () => ok(null);
  });
};
const idbPut = async (key, value) => {
  const db = await IDB();
  db.transaction("reflow", "readwrite").objectStore("reflow").put(value, key);
};

/** Pictures on a page, from its drawing commands: where each sits, and how big. */
async function pageImages(page, pageW, pageH) {
  const ops = await page.getOperatorList();
  const O = pdfjs.OPS;
  const stack = [];
  let m = [1, 0, 0, 1, 0, 0];
  const out = [];
  for (let i = 0; i < ops.fnArray.length; i++) {
    const fn = ops.fnArray[i], args = ops.argsArray[i];
    if (fn === O.save) stack.push(m);
    else if (fn === O.restore) m = stack.pop() ?? m;
    else if (fn === O.transform) {
      const [a, b, c, d, e, f] = args;
      m = [m[0] * a + m[2] * b, m[1] * a + m[3] * b, m[0] * c + m[2] * d, m[1] * c + m[3] * d, m[0] * e + m[2] * f + m[4], m[1] * e + m[3] * f + m[5]];
    } else if (fn === O.paintImageXObject) {
      const w = Math.hypot(m[0], m[1]), h = Math.hypot(m[2], m[3]);
      if (w >= pageW * 0.12 && h >= pageH * 0.06) out.push({ name: args[0], w, h, y: m[5] + h });
    }
  }
  return out;
}

const toJpeg = (canvas, q = 0.82) => new Promise((ok) => canvas.toBlob(ok, "image/jpeg", q));

async function imageBlob(page, name) {
  const v = await imageObject(page, name);
  if (!v) return null;
  const max = 1400;
  const scale = Math.min(1, max / Math.max(v.width, v.height));
  const c = document.createElement("canvas");
  c.width = Math.round(v.width * scale);
  c.height = Math.round(v.height * scale);
  const x = c.getContext("2d");
  x.fillStyle = "#fff";
  x.fillRect(0, 0, c.width, c.height);
  if (v.bitmap) x.drawImage(v.bitmap, 0, 0, c.width, c.height);
  else if (v.data) {
    const src = document.createElement("canvas");
    src.width = v.width;
    src.height = v.height;
    const data = new ImageData(v.width, v.height);
    const n = v.width * v.height;
    if (v.data.length === n * 4) data.data.set(v.data);
    else if (v.data.length === n * 3) for (let i = 0; i < n; i++) { data.data[i * 4] = v.data[i * 3]; data.data[i * 4 + 1] = v.data[i * 3 + 1]; data.data[i * 4 + 2] = v.data[i * 3 + 2]; data.data[i * 4 + 3] = 255; }
    else return null;
    src.getContext("2d").putImageData(data, 0, 0);
    x.drawImage(src, 0, 0, c.width, c.height);
  } else return null;
  return { blob: await toJpeg(c), width: c.width, height: c.height };
}

/** A whole page as a picture: a plate, or a page that has no text. */
async function pageBlob(page) {
  const base = page.getViewport({ scale: 1 });
  const scale = Math.min(2, 1100 / base.width);
  const vp = page.getViewport({ scale });
  const c = document.createElement("canvas");
  c.width = vp.width;
  c.height = vp.height;
  await page.render({ canvas: c, canvasContext: c.getContext("2d"), viewport: vp }).promise;
  if (looksBlank(c)) return null; // a blank leaf: nothing to read, and a glaring white block at night
  return { blob: await toJpeg(c, 0.75), width: c.width, height: c.height, scan: true };
}

async function pageText(page) {
  const tc = await page.getTextContent();
  return tc.items
    .filter((i) => i.str !== undefined)
    .map((i) => ({ str: i.str, x: i.transform[4], y: i.transform[5], w: i.width, h: Math.hypot(i.transform[2], i.transform[3]) || i.height }));
}

async function extractPDF(cacheKey) {
  const gen = S.gen;
  const N = S.doc.numPages;

  // a few pages say which kind of PDF this is
  const sample = [];
  for (const n of [...new Set([0.1, 0.3, 0.5, 0.7, 0.9].map((f) => Math.max(1, Math.round(N * f))))]) {
    const page = await S.doc.getPage(n);
    const vp = page.getViewport({ scale: 1 });
    const text = await pageText(page);
    const pics = await pageImages(page, vp.width, vp.height);
    sample.push({ chars: text.reduce((a, i) => a + i.str.length, 0), fullImage: pics.some((p) => p.w * p.h > vp.width * vp.height * 0.7) });
  }
  const type = classify(sample);
  log("info", "pdf kind", { type, pages: N, sample });
  if (type === "scanned") return { type, blocks: [], pageStart: {} };

  const t0 = performance.now();
  const pages = [];
  for (let n = 1; n <= N; n++) {
    if (gen !== S.gen) return null;
    if (n % 4 === 0) {
      prep("Preparing reading mode", n / N);
      await breathe();
    }
    const page = await S.doc.getPage(n);
    const vp = page.getViewport({ scale: 1 });
    const items = await pageText(page);
    const chars = items.reduce((a, i) => a + i.str.length, 0);
    const images = [];
    // OCR scans have a picture of the whole page under the text: only keep it when
    // there's little text, which means a plate or a title page
    if (type === "digital" || chars < 200) {
      const pics = await pageImages(page, vp.width, vp.height);
      const full = pics.some((p) => p.w * p.h > vp.width * vp.height * 0.7);
      if (full || (chars < 40 && pics.length)) {
        if (chars < 200) {
          const whole = await pageBlob(page);
          if (whole) images.push({ ...whole, y: vp.height });
        }
      } else {
        for (const p of pics.slice(0, 8)) {
          const b = await imageBlob(page, p.name);
          if (b) images.push({ ...b, y: p.y });
        }
      }
    }
    const lines = chars >= 200 || !images.length ? orderColumns(buildLines(items), vp.width) : [];
    pages.push({ n, lines, images });
    page.cleanup();
  }
  const { blocks, pageStart } = toBlocks(stripRunning(pages), { noisy: type === "ocr" });
  const out = { type, blocks, pageStart };
  log("info", "pdf reflowed", { ms: Math.round(performance.now() - t0), type, pages: N, blocks: blocks.length, images: blocks.filter((b) => b.tag === "img").length });
  try {
    await idbPut(cacheKey, out);
  } catch {
    // no room: it's worked out again next time
  }
  return out;
}

async function pdfFlow(cacheKey) {
  let data = null;
  try {
    data = await idbGet(cacheKey);
  } catch {
    // no stored copy, then
  }
  if (!data) data = await extractPDF(cacheKey);
  if (!data) return false;
  if (data.type === "scanned" || !data.blocks.length) {
    send({ type: "scanned" });
    return false;
  }
  const blocks = data.blocks.map((b) => (b.tag === "img" ? { ...b, src: URL.createObjectURL(b.blob) } : b));
  const { sections, pageSection } = toSections(blocks, data.pageStart);
  S.sections = sections;
  S.pdfSection = pageSection;
  return true;
}

// ---- EPUB

const KEEP = new Set("p div h1 h2 h3 h4 h5 h6 blockquote ul ol li em i strong b u s sub sup small span br hr img table thead tbody tfoot tr td th caption pre code section article figure figcaption aside dl dt dd abbr cite q center".split(" "));
const DROP = new Set("script style head title link meta iframe object embed form input button select textarea audio video canvas math".split(" "));
const IMG_TYPES = { jpg: "image/jpeg", jpeg: "image/jpeg", png: "image/png", gif: "image/gif", webp: "image/webp", svg: "image/svg+xml" };

/** Only the book's words and pictures come through: no scripts, styles or links. */
function cleanInto(target, node, image) {
  for (const n of node.childNodes) {
    if (n.nodeType === 3) {
      target.append(n.textContent);
      continue;
    }
    if (n.nodeType !== 1) continue;
    const tag = n.localName.toLowerCase();
    if (DROP.has(tag)) continue;
    if (tag === "img" || tag === "image") {
      const pic = image(n.getAttribute("src") ?? n.getAttribute("href") ?? n.getAttribute("xlink:href") ?? n.getAttributeNS?.("http://www.w3.org/1999/xlink", "href"));
      if (pic) {
        const el = document.createElement("img");
        el.src = pic.url;
        if (pic.width) { el.width = pic.width; el.height = pic.height; }
        el.alt = "";
        target.append(el);
      }
      continue;
    }
    // ids stay as anchors: the contents point at them
    const id = n.getAttribute("id");
    if (!KEEP.has(tag)) {
      if (id) {
        const a = document.createElement("span");
        a.setAttribute("data-a", id);
        target.append(a);
      }
      cleanInto(target, n, image);
      continue;
    }
    const el = document.createElement(tag);
    if (id) el.setAttribute("data-a", id);
    cleanInto(el, n, image);
    if (tag === "td" || tag === "th") for (const a of ["colspan", "rowspan"]) if (n.hasAttribute(a)) el.setAttribute(a, n.getAttribute(a));
    if (el.childNodes.length || id || tag === "br" || tag === "hr") target.append(el);
  }
}

/** Cuts a chapter's HTML into pieces short enough to lay out quickly. */
function piecesOf(root, limit = 40000) {
  let box = root;
  const wrapper = new Set(["div", "section", "article"]);
  while (box.children.length === 1 && wrapper.has(box.children[0].localName) && !box.textContent.replace(box.children[0].textContent, "").trim()) box = box.children[0];
  const out = [];
  let cur = "";
  for (const n of box.childNodes) {
    const html = n.nodeType === 1 ? n.outerHTML : n.textContent.trim() ? `<p>${n.textContent.replace(/[&<>]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;" })[c])}</p>` : "";
    if (cur && cur.length + html.length > limit) {
      out.push(cur);
      cur = "";
    }
    cur += html;
  }
  if (cur) out.push(cur);
  return out;
}

const parseXML = (text) => new DOMParser().parseFromString(text, "application/xml");
const childrenBy = (doc, name) => [...doc.getElementsByTagNameNS("*", name)];
const relative = (base, href) => {
  const u = new URL(href, `http://x/${base}`);
  return decodeURIComponent(u.pathname.slice(1));
};

async function openEPUB(url) {
  const zip = window.fflate.unzipSync(await readBytes(url), { filter: (f) => f.originalSize < 150 * 1024 * 1024 });
  const decoder = new TextDecoder();
  const text = (p) => (zip[p] ? decoder.decode(zip[p]) : "");
  const container = parseXML(text("META-INF/container.xml"));
  const opfPath = childrenBy(container, "rootfile")[0]?.getAttribute("full-path");
  if (!opfPath) throw new Error("not a readable EPUB");
  const opf = parseXML(text(opfPath));
  S.lang = childrenBy(opf, "language")[0]?.textContent?.slice(0, 2) || "en";
  document.documentElement.lang = S.lang;

  const manifest = new Map();
  for (const it of childrenBy(opf, "item")) {
    manifest.set(it.getAttribute("id"), { path: relative(opfPath, it.getAttribute("href")), type: it.getAttribute("media-type") ?? "", props: it.getAttribute("properties") ?? "" });
  }
  const spine = childrenBy(opf, "itemref").map((r) => manifest.get(r.getAttribute("idref"))).filter(Boolean);

  // pictures first: their sizes have to be known to lay pages out
  const pics = new Map();
  for (const m of manifest.values()) {
    const ext = m.path.split(".").pop().toLowerCase();
    const type = IMG_TYPES[ext];
    if (!type || !zip[m.path]) continue;
    const blob = new Blob([zip[m.path]], { type });
    let width = 0, height = 0;
    if (type !== "image/svg+xml") {
      try {
        const bmp = await createImageBitmap(blob);
        width = bmp.width;
        height = bmp.height;
        bmp.close();
      } catch {
        continue;
      }
    }
    pics.set(m.path, { url: URL.createObjectURL(blob), width, height, blob, type });
  }

  const sections = [];
  const firstSection = new Map();
  for (let i = 0; i < spine.length; i++) {
    const m = spine[i];
    if (i % 5 === 0) {
      prep("Opening the book", i / spine.length);
      await breathe();
    }
    let html = text(m.path);
    let doc = new DOMParser().parseFromString(html, "application/xhtml+xml");
    if (doc.querySelector("parsererror")) doc = new DOMParser().parseFromString(html, "text/html");
    const body = doc.getElementsByTagName("body")[0] ?? doc.documentElement;
    const root = document.createElement("div");
    cleanInto(root, body, (src) => (src ? pics.get(relative(m.path, src)) : null));
    html = null;
    const parts = piecesOf(root);
    if (!parts.length) continue;
    firstSection.set(m.path, [sections.length, sections.length + parts.length]);
    sections.push(...parts);
  }
  S.sections = sections;

  // the table of contents: EPUB 3's nav, or the older NCX
  const toc = [];
  const add = (title, base, href, depth) => {
    const [path, anchor] = href.split("#");
    const range = firstSection.get(relative(base, path));
    if (!title?.trim() || !range) return;
    // a long chapter file is cut into pieces: find the piece the anchor's in
    let section = range[0];
    if (anchor) for (let i = range[0]; i < range[1]; i++) if (sections[i].includes(`data-a="${anchor}"`)) section = i;
    toc.push({ title: tidyTitle(title), section, depth, anchor });
  };
  const levels = (el, name) => {
    let d = -1;
    for (let n = el; n; n = n.parentElement) if (n.localName === name) d++;
    return Math.max(0, d);
  };
  const nav = [...manifest.values()].find((m) => m.props.split(" ").includes("nav"));
  const ncx = [...manifest.values()].find((m) => m.type === "application/x-dtbncx+xml");
  if (nav) {
    const d = new DOMParser().parseFromString(text(nav.path), "application/xhtml+xml");
    const list = [...d.getElementsByTagName("nav")].find((n) => (n.getAttribute("epub:type") ?? "toc") === "toc") ?? d;
    for (const a of list.getElementsByTagName("a")) add(a.textContent, nav.path, a.getAttribute("href") ?? "", levels(a, "ol"));
  } else if (ncx) {
    const d = parseXML(text(ncx.path));
    for (const p of childrenBy(d, "navPoint")) add(childrenBy(p, "text")[0]?.textContent, ncx.path, childrenBy(p, "content")[0]?.getAttribute("src") ?? "", levels(p, "navPoint"));
  }
  S.toc = toc.slice(0, 400);

  // its cover, for the shelf
  const coverId = childrenBy(opf, "meta").find((x) => x.getAttribute("name") === "cover")?.getAttribute("content");
  const coverItem = [...manifest.values()].find((m) => m.props.split(" ").includes("cover-image")) ?? manifest.get(coverId);
  return coverItem ? pics.get(coverItem.path) ?? null : null;
}

// ---- Markdown

function openMarkdown(text) {
  const renderer = new window.marked.Renderer();
  const esc = (s) => s.replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" })[c]);
  renderer.html = ({ text: t }) => esc(t);
  renderer.link = ({ text: t }) => esc(t);
  renderer.image = ({ text: t }) => esc(t ?? "");
  const box = document.createElement("div");
  box.innerHTML = window.marked.parse(text, { renderer, async: false, gfm: true });
  S.sections = piecesOf(box);
}

// ---- laying out pages

const pageW = () => W() - S.margin * 2;
const head = () => 34 + S.safeTop * 0.4;
const foot = () => 40 + S.safeBottom;
const pageH = () => H() - head() - foot();

function flowEl(html) {
  const el = document.createElement("div");
  el.className = "flow";
  el.innerHTML = html;
  el.style.cssText = `left:${S.margin}px;top:${head()}px;width:${pageW()}px;height:${pageH()}px;column-width:${pageW()}px;column-gap:${S.margin * 2}px;--size:${S.size}px;--leading:${S.leading};--font:${S.font};--align:${S.align};--pageh:${pageH() - 24}px`;
  return el;
}

async function layoutAll(label) {
  const probe = flowEl("");
  probe.style.visibility = "hidden";
  document.body.append(probe);
  const step = pageW() + S.margin * 2;
  S.counts = [];
  const gen = S.gen;
  for (let i = 0; i < S.sections.length; i++) {
    if (i % 6 === 0 && S.sections.length > 12) {
      prep(label, i / S.sections.length);
      await breathe();
      if (gen !== S.gen) { probe.remove(); return; }
    }
    probe.innerHTML = S.sections[i];
    S.counts.push(Math.max(1, Math.round((probe.scrollWidth + S.margin * 2) / step)));
  }
  probe.remove();
  S.starts = [];
  let at = 1;
  for (const c of S.counts) {
    S.starts.push(at);
    at += c;
  }
  S.total = Math.max(1, at - 1);
}

function sectionOf(n) {
  let lo = 0, hi = S.starts.length - 1;
  while (lo < hi) {
    const mid = Math.ceil((lo + hi) / 2);
    if (S.starts[mid] <= n) lo = mid;
    else hi = mid - 1;
  }
  return lo;
}

function flowPage(n) {
  const i = sectionOf(n);
  const el = flowEl(S.sections[i] ?? "");
  el.dataset.section = String(i);
  if (S.mark?.kind === "flow" && S.mark.section === i) {
    const p = el.children[S.mark.block];
    if (p) {
      p.classList.add("resume");
      if (S.pulse) p.classList.add("pulse");
    }
  }
  el.style.transform = `translateX(${-(n - S.starts[i]) * (pageW() + S.margin * 2)}px)`;
  const clip = document.createElement("div");
  clip.className = "clip";
  clip.append(el);
  return clip;
}

function sheetFor(w, h) {
  const sheet = document.createElement("div");
  sheet.className = "sheet";
  sheet.style.width = `${w}px`;
  sheet.style.height = `${h}px`;
  return sheet;
}

async function draw(leaf, n) {
  const gen = S.gen;
  let sheet;
  if (isFlow()) {
    sheet = sheetFor(W(), H());
    sheet.append(flowPage(n));
    const folio = document.createElement("div");
    folio.id = "folio";
    folio.textContent = String(n);
    folio.style.bottom = `${12 + S.safeBottom}px`;
    sheet.append(folio);
  } else if (S.kind === "pdf") {
    const r = await renderPDF(n);
    if (gen !== S.gen) return;
    sheet = sheetFor(r.w, r.h);
    sheet.append(r.canvas.cloneNode(false));
    sheet.firstChild.getContext("2d").drawImage(r.canvas, 0, 0);
    sheet.append(r.layer.cloneNode(true));
    if (S.mark?.kind === "page" && S.mark.page === n) {
      const rib = document.createElement("div");
      rib.className = `ribbon${S.pulse ? " pulse" : ""}`;
      rib.style.top = `${Math.round(S.mark.y * r.h)}px`;
      sheet.append(rib);
    }
  } else {
    sheet = sheetFor(W(), H());
    const img = document.createElement("img");
    img.className = "pic";
    img.src = S.url;
    img.draggable = false;
    sheet.append(img);
  }
  const shade = document.createElement("div");
  shade.className = "shade";
  leaf.replaceChildren(sheet, shade);
  applyZoom(leaf);
}

// ---- the resume mark

/** The paragraph (or the spot on an original page) under a finger. */
function targetAt(x, y) {
  const el = document.elementFromPoint(x, y);
  if (isFlow()) {
    const flow = el?.closest?.(".flow");
    if (!flow) return null;
    let n = el;
    while (n && n.parentElement !== flow) n = n.parentElement;
    if (!n || n === flow) return null;
    const block = [...flow.children].indexOf(n);
    return { kind: "flow", section: Number(flow.dataset.section), block, snippet: n.textContent.replace(/\s+/g, " ").trim().slice(0, 110) };
  }
  if (S.kind === "pdf") {
    const sheet = top.querySelector(".sheet");
    if (!sheet) return null;
    const r = sheet.getBoundingClientRect();
    if (y < r.top || y > r.bottom) return null;
    const near = wordAt(x, y)?.context ?? "";
    return { kind: "page", page: S.page, y: Math.max(0, (y - r.top) / r.height - 0.012), snippet: near.slice(0, 110) };
  }
  return null;
}

const sameMark = (a, b) => !!a && !!b && a.kind === b.kind && (a.kind === "flow" ? a.section === b.section && a.block === b.block : a.page === b.page && Math.abs(a.y - b.y) < 0.04);

/** The page a mark is on, in the current layout; null if this layout can't show it. */
function markPage(m) {
  if (!m) return null;
  if (m.kind === "page") return S.kind === "pdf" && !isFlow() && m.page <= S.total ? m.page : null;
  if (!isFlow() || m.section >= S.sections.length) return null;
  const probe = flowEl(S.sections[m.section]);
  probe.style.visibility = "hidden";
  document.body.append(probe);
  const p = probe.children[m.block];
  const step = pageW() + S.margin * 2;
  const at = p ? Math.floor((p.offsetLeft + 2) / step) : 0;
  probe.remove();
  return Math.min(S.total, S.starts[m.section] + Math.max(0, at));
}

async function setMark(m) {
  S.mark = m;
  S.pulse = false;
  log("info", m ? "mark set" : "mark removed", m ? { kind: m.kind, page: S.page } : undefined);
  await draw(top, S.page);
}

async function goToMark() {
  const p = markPage(S.mark);
  if (p == null) return send({ type: "markLost" });
  S.pulse = true;
  if (p === S.page) await draw(top, p);
  else await goTo(p);
  setTimeout(() => (S.pulse = false), 3500);
}

// ---- opening

/** Is this page nearly all one colour? Scanned books often open on a blank flyleaf. */
function looksBlank(canvas) {
  const small = document.createElement("canvas");
  small.width = 32;
  small.height = 46;
  const x = small.getContext("2d");
  x.drawImage(canvas, 0, 0, 32, 46);
  const d = x.getImageData(0, 0, 32, 46).data;
  const first = (d[0] + d[1] + d[2]) / 3;
  let far = 0;
  for (let i = 0; i < d.length; i += 4) if (Math.abs((d[i] + d[i + 1] + d[i + 2]) / 3 - first) > 24) far++;
  return far < (d.length / 4) * 0.015;
}

async function coverFrom(source, width = 320) {
  const c = document.createElement("canvas");
  if (source.doc) {
    // the first page with something on it
    for (let n = 1; n <= Math.min(8, source.doc.numPages); n++) {
      const page = await source.doc.getPage(n);
      const base = page.getViewport({ scale: 1 });
      const vp = page.getViewport({ scale: width / base.width });
      c.width = vp.width;
      c.height = vp.height;
      await page.render({ canvas: c, canvasContext: c.getContext("2d"), viewport: vp }).promise;
      if (!looksBlank(c)) break;
      log("debug", "cover: blank page skipped", { n });
    }
  } else {
    const bmp = await createImageBitmap(source.blob);
    c.width = width;
    c.height = Math.round((bmp.height / bmp.width) * width);
    c.getContext("2d").drawImage(bmp, 0, 0, c.width, c.height);
    bmp.close();
  }
  send({ type: "cover", data: c.toDataURL("image/jpeg", 0.8).split(",")[1] });
}

/** A PDF with no list of chapters: read the pages for headings, once, in the background. */
async function findChapters(cacheKey) {
  const gen = S.gen;
  const key = `${cacheKey}:toc`;
  let found = null;
  try {
    found = await idbGet(key);
  } catch {
    // worked out again, then
  }
  if (!found) {
    send({ type: "tocBusy", busy: true });
    const pages = [];
    const N = Math.min(S.doc.numPages, 3000);
    for (let n = 1; n <= N; n++) {
      if (gen !== S.gen) return;
      const page = await S.doc.getPage(n);
      pages.push({ n, lines: buildLines(await pageText(page)) });
      page.cleanup();
      if (n % 6 === 0) await breathe();
    }
    found = guessContents(pages);
    log("info", "chapters found", { pages: N, chapters: found.length });
    idbPut(key, found).catch(() => undefined);
  }
  if (gen !== S.gen) return;
  S.toc = found;
  send({ type: "tocBusy", busy: false });
  sendToc();
  refreshToc().catch(() => undefined);
}

const outline = async () => {
  const out = [];
  const walk = async (items, depth) => {
    for (const it of items ?? []) {
      if (out.length > 400) return;
      try {
        const dest = typeof it.dest === "string" ? await S.doc.getDestination(it.dest) : it.dest;
        if (dest) out.push({ title: it.title, pdfPage: (await S.doc.getPageIndex(dest[0])) + 1, depth });
      } catch {
        // a link that goes nowhere
      }
      if (depth < 3) await walk(it.items, depth + 1);
    }
  };
  await walk(await S.doc.getOutline(), 0);
  return out;
};

/** Exact pages for the contents: where each anchor falls. In the background, a piece at a time. */
async function refreshToc() {
  if (!isFlow()) return;
  const gen = S.gen;
  const bySection = new Map();
  for (const c of S.toc) {
    c.offset = 0;
    const section = c.pdfPage !== undefined ? S.pdfSection?.[c.pdfPage] : c.section;
    const anchor = c.pdfPage !== undefined ? `p${c.pdfPage}` : c.anchor;
    if (section === undefined || !anchor) continue;
    if (!bySection.has(section)) bySection.set(section, []);
    bySection.get(section).push([c, anchor]);
  }
  if (!bySection.size) return;
  const probe = flowEl("");
  probe.style.visibility = "hidden";
  document.body.append(probe);
  const step = pageW() + S.margin * 2;
  for (const [section, list] of bySection) {
    if (gen !== S.gen) break;
    probe.innerHTML = S.sections[section] ?? "";
    for (const [c, anchor] of list) {
      const el = probe.querySelector(`[data-a="${CSS.escape(anchor)}"]`);
      // an empty marker can be left at the foot of a page, with the chapter over the leaf
      const at = el && !el.textContent.trim() ? (el.nextElementSibling ?? el) : el;
      if (at) c.offset = Math.max(0, Math.floor((at.offsetLeft + 2) / step));
    }
    await breathe();
  }
  probe.remove();
  if (gen === S.gen) sendToc();
}

/** The contents, in this layout's page numbers. */
const sendToc = () => {
  const toc = S.toc
    .map((c) => {
      if (c.pdfPage !== undefined) {
        if (!isFlow()) return { title: c.title, page: c.pdfPage, depth: c.depth ?? 0 };
        const sec = S.pdfSection?.[c.pdfPage];
        return sec === undefined ? null : { title: c.title, page: (S.starts[sec] ?? 1) + (c.offset ?? 0), depth: c.depth ?? 0 };
      }
      return { title: c.title, page: (S.starts[c.section] ?? 1) + (c.offset ?? 0), depth: c.depth ?? 0 };
    })
    .filter(Boolean);
  send({ type: "toc", toc });
};

async function open({ id, url, kind, page, savedTotal, theme, settings, text, layout, cacheKey, wantCover, mark }) {
  S.gen++;
  const t0 = performance.now();
  log("info", "open", { kind, layout, page, savedTotal, mark: mark?.kind ?? null });
  cache.clear();
  Object.assign(S, { id, kind, url, layout: layout || "text", zoom: 1, panX: 0, panY: 0, doc: null, sections: [], toc: [], pdfSection: null });
  Object.assign(S, settings ?? {});
  S.mark = mark ?? null;
  S.pulse = false;
  setTheme(theme || "night");
  try {
    let coverSource = null;
    if (kind === "pdf") {
      prep("Opening the book", 0);
      const here = (p) => new URL(p, location.href).href;
      S.doc = await pdfjs.getDocument({
        data: await readBytes(url),
        isEvalSupported: false,
        enableXfa: false,
        useSystemFonts: true,
        wasmUrl: here("pdf/wasm/"),
        cMapUrl: here("pdf/cmaps/"),
        cMapPacked: true,
        standardFontDataUrl: here("pdf/standard_fonts/"),
        iccUrl: here("pdf/iccs/"),
      }).promise;
      S.total = S.doc.numPages;
      S.toc = await outline().catch(() => []);
      if (wantCover) coverFrom({ doc: S.doc }).catch((e) => log("warn", "cover failed", { message: String(e?.message ?? e) }));
      if (S.layout === "text" && !(await pdfFlow(cacheKey))) S.layout = "page";
      send({ type: "layout", layout: S.layout });
    } else if (kind === "epub") {
      prep("Opening the book", 0);
      coverSource = await openEPUB(url);
    } else if (kind === "markdown") {
      openMarkdown(text ?? "");
    } else {
      S.total = 1;
    }
    if (isFlow()) await layoutAll("Setting the pages");
    S.page = rescale(page, savedTotal, S.total);
    // a resume mark wins: that's where you said you'd pick up
    const resume = markPage(S.mark);
    if (resume != null) {
      S.page = resume;
      S.pulse = true;
      setTimeout(() => (S.pulse = false), 3500);
    }
    await draw(top, S.page);
    prep(null);
    log("info", "opened", { kind, layout: S.layout, pages: S.total, sections: S.sections.length, page: S.page, resumed: resume != null, ms: Math.round(performance.now() - t0) });
    send({ type: "loaded", page: S.page, total: S.total, flow: isFlow(), resumed: resume != null });
    sendToc();
    refreshToc().catch(() => undefined);
    if (kind === "pdf" && !S.toc.length) findChapters(cacheKey).catch((e) => log("warn", "finding chapters failed", { message: String(e?.message ?? e) }));
    if (wantCover && coverSource) coverFrom(coverSource).catch(() => undefined);
    prerender();
  } catch (e) {
    prep(null);
    log("error", "open failed", { kind, message: String(e?.message ?? e), stack: String(e?.stack ?? "").slice(0, 600) });
    top.innerHTML = `<div class="err">This file can't be read.<br><small></small></div>`;
    top.querySelector("small").textContent = String(e?.message ?? e);
    send({ type: "error", message: String(e?.message ?? e) });
  }
}

const prerender = () => {
  if (S.kind !== "pdf" || isFlow()) return;
  for (const n of [S.page + 1, S.page - 1]) if (n >= 1 && n <= S.total) renderPDF(n).catch(() => undefined);
};

function setTheme(theme) {
  document.body.className = theme;
  send({ type: "theme", theme });
}

/** Text settings moved the pages: stays at the same place in the book. */
async function relayout() {
  const at = (S.page - 1) / Math.max(1, S.total - 1);
  await layoutAll("Setting the pages");
  S.page = Math.max(1, Math.round(at * (S.total - 1)) + 1);
  await draw(top, S.page);
  prep(null);
  send({ type: "page", page: S.page, total: S.total });
  sendToc();
  refreshToc().catch(() => undefined);
}

async function setSettings(next) {
  const before = `${S.size}|${S.leading}|${S.margin}|${S.font}|${S.align}`;
  Object.assign(S, next);
  if (isFlow() && before !== `${S.size}|${S.leading}|${S.margin}|${S.font}|${S.align}`) await relayout();
}

async function setLayout(layout, cacheKey) {
  if (S.kind !== "pdf" || layout === S.layout) return;
  const at = (S.page - 1) / Math.max(1, S.total - 1);
  S.layout = layout;
  S.zoom = 1;
  if (layout === "text") {
    prep("Preparing reading mode", 0);
    await breathe();
    if (!S.sections.length && !(await pdfFlow(cacheKey))) S.layout = "page";
    else await layoutAll("Setting the pages");
  }
  if (S.layout === "page") S.total = S.doc.numPages;
  S.page = Math.max(1, Math.round(at * (S.total - 1)) + 1);
  await draw(top, S.page);
  prep(null);
  send({ type: "layout", layout: S.layout });
  send({ type: "page", page: S.page, total: S.total, flow: isFlow() });
  sendToc();
  refreshToc().catch(() => undefined);
  prerender();
}

// ---- turning

// slow to lift, quick through the middle, settling down: a hand, not a machine
const curve = (t) => (t < 0.5 ? 4 * t * t * t : 1 - (-2 * t + 2) ** 3 / 2);

function setTurn(leaf, p) {
  const q = Math.min(Math.max(p, 0), 1);
  leaf.style.transform = q ? `rotateY(${-q * 100}deg)` : "";
  const shade = leaf.querySelector(".shade");
  if (shade) shade.style.opacity = String(Math.min(1, q * 2.4));
  cast.style.opacity = String(q > 0 && q < 1 ? Math.sin(q * Math.PI) : 0);
}

function animate(leaf, from, to, ms) {
  return new Promise((done) => {
    const t0 = performance.now();
    const step = (now) => {
      const t = Math.min(1, (now - t0) / Math.max(ms, 1));
      setTurn(leaf, from + (to - from) * curve(t));
      if (t < 1) requestAnimationFrame(step);
      else done();
    };
    requestAnimationFrame(step);
  });
}

const flipMs = () => (S.turn === "fold" ? S.speed * 0.55 : S.speed);

async function prepare(dir) {
  const n = S.page + dir;
  if (n < 1 || n > S.total) return false;
  if (dir > 0) {
    await draw(under, n);
  } else {
    // going back, the earlier page comes down from the spine, over this one
    under.replaceChildren(...top.childNodes);
    await draw(top, n);
    setTurn(top, 1);
  }
  return true;
}

function settle(dir, turned) {
  if (turned) {
    S.page += dir;
    S.scrollY = 0;
    if (dir > 0) top.replaceChildren(...under.childNodes);
    send({ type: "page", page: S.page, total: S.total });
    prerender();
  } else if (dir < 0) top.replaceChildren(...under.childNodes);
  setTurn(top, 0);
  under.replaceChildren();
  S.turning = false;
}

async function turn(dir) {
  if (S.turning) return;
  if (S.zoom > 1) return;
  // a page taller than the screen: read down it before turning
  const room = maxScroll();
  if (room > 0 && ((dir > 0 && S.scrollY < room - 1) || (dir < 0 && S.scrollY > 1))) {
    S.scrollY = Math.min(room, Math.max(0, S.scrollY + dir * H() * 0.85));
    applyZoom();
    return;
  }
  const n = S.page + dir;
  if (n < 1 || n > S.total) return send({ type: "edge", dir });
  S.turning = true;
  if (S.turn === "none") {
    S.page = n;
    S.scrollY = 0;
    await draw(top, n);
    S.turning = false;
    send({ type: "page", page: n, total: S.total });
    prerender();
    return;
  }
  await prepare(dir);
  if (dir > 0) await animate(top, 0, 1, flipMs());
  else await animate(top, 1, 0, flipMs());
  settle(dir, true);
}

async function goTo(n) {
  n = Math.min(Math.max(n, 1), S.total);
  if (n === S.page) return;
  S.page = n;
  S.zoom = 1;
  S.scrollY = 0;
  await draw(top, n);
  send({ type: "page", page: n, total: S.total });
  prerender();
}

// ---- zoom, on original pages

const sheetH = () => Number.parseInt(top.firstChild?.style?.height ?? "0", 10) || 0;
const maxScroll = () => (isFlow() ? 0 : Math.max(0, sheetH() - H()));

function applyZoom(leaf = top) {
  const s = leaf.firstChild;
  if (!s?.style) return;
  // a sheet taller than the screen sits centred: move it so its top shows, then by how far you've read
  const h = Number.parseInt(s.style.height, 10) || 0;
  const scroll = h > H() ? `translateY(${(h - H()) / 2 - (leaf === top ? S.scrollY : 0)}px)` : "";
  s.style.transform = S.zoom > 1 ? `translate(${S.panX}px, ${S.panY}px) scale(${S.zoom})` : scroll;
}

function toggleZoom(x, y) {
  if (S.zoom > 1) {
    S.zoom = 1;
    S.panX = S.panY = 0;
  } else {
    S.zoom = 2.2;
    S.panX = (W() / 2 - x) * (S.zoom - 1);
    S.panY = (H() / 2 - y) * (S.zoom - 1);
  }
  applyZoom();
}

// ---- looking up a word

function wordAt(x, y) {
  const r = document.caretRangeFromPoint?.(x, y);
  const node = r?.startContainer;
  if (!node || node.nodeType !== 3) return null;
  const t = node.textContent;
  const isWord = (c) => c && /[\p{L}\p{M}'’-]/u.test(c);
  let a = r.startOffset, b = r.startOffset;
  while (a > 0 && isWord(t[a - 1])) a--;
  while (b < t.length && isWord(t[b])) b++;
  const word = t.slice(a, b).replace(/^['’-]+|['’-]+$/g, "");
  if (!word) return null;
  const range = document.createRange();
  range.setStart(node, a);
  range.setEnd(node, b);
  return { word, rect: range.getBoundingClientRect(), context: t.slice(Math.max(0, a - 60), Math.min(t.length, b + 60)).trim() };
}

function mark(rect) {
  const m = $("mark");
  if (!rect) { m.style.display = "none"; return; }
  Object.assign(m.style, { display: "block", left: `${rect.left - 2}px`, top: `${rect.top - 1}px`, width: `${rect.width + 4}px`, height: `${rect.height + 2}px` });
}

// ---- the finger

let P = null;
let lastTap = 0;
let tapTimer = 0;

addEventListener("pointerdown", (e) => {
  mark(null);
  P = { x: e.clientX, y: e.clientY, x0: e.clientX, y0: e.clientY, t: performance.now(), moved: false, dir: 0, ready: false, pending: null };
  P.hold = setTimeout(() => {
    if (!P || P.moved) return;
    P.held = true;
    const w = wordAt(P.x, P.y);
    const target = targetAt(P.x, P.y);
    if (w) mark(w.rect);
    if (w || target) send({ type: "hold", word: w?.word ?? null, context: w?.context ?? "", target, marked: sameMark(target, S.mark) });
  }, 430);
});

addEventListener("pointermove", (e) => {
  if (!P) return;
  const dx = e.clientX - P.x0, dy = e.clientY - P.y0;
  if (!P.moved && Math.hypot(dx, dy) > 10) {
    P.moved = true;
    clearTimeout(P.hold);
  }
  if (!P.moved || P.held) return;
  if (S.zoom > 1) {
    S.panX += e.clientX - P.x;
    S.panY += e.clientY - P.y;
    P.x = e.clientX;
    P.y = e.clientY;
    applyZoom();
    return;
  }
  P.x = e.clientX;
  // up and down moves a page taller than the screen
  if (!P.dir && maxScroll() > 0 && (P.scrolling || Math.abs(dy) > Math.abs(dx))) {
    if (!P.scrolling) {
      P.scrolling = true;
      P.from = S.scrollY;
    }
    S.scrollY = Math.min(maxScroll(), Math.max(0, P.from - dy));
    applyZoom();
    return;
  }
  // the page follows the finger, once a direction is chosen
  if (!P.dir && !S.turning && S.turn !== "none" && Math.abs(dx) > 14 && Math.abs(dx) > Math.abs(dy)) {
    P.dir = dx < 0 ? 1 : -1;
    S.turning = true;
    P.pending = prepare(P.dir).then((ok) => {
      if (!ok) {
        S.turning = false;
        send({ type: "edge", dir: P?.dir ?? 0 });
        if (P) P.dir = 0;
        return;
      }
      if (P) P.ready = true;
    });
  }
  if (P.ready) {
    const raw = Math.abs(dx) / (W() * 0.85);
    setTurn(top, P.dir > 0 ? Math.min(raw, 1) : Math.max(1 - raw, 0));
  }
});

addEventListener("pointerup", async (e) => {
  const p = P;
  P = null;
  if (!p) return;
  clearTimeout(p.hold);
  if (p.held) return;
  const dx = e.clientX - p.x0;
  if (p.dir) {
    await p.pending;
    if (!p.ready) return;
    const progress = Math.min(Math.abs(dx) / (W() * 0.85), 1);
    const flick = performance.now() - p.t < 280 && Math.abs(dx) > 30;
    const done = progress > 0.28 || flick;
    const rest = Math.max(90, flipMs() * 0.55);
    if (p.dir > 0) await animate(top, progress, done ? 1 : 0, rest);
    else await animate(top, 1 - progress, done ? 0 : 1, rest);
    settle(p.dir, done);
    return;
  }
  // with page turns off, a swipe still turns
  if (p.moved && S.turn === "none" && S.zoom === 1 && Math.abs(dx) > 40) return turn(dx < 0 ? 1 : -1);
  if (p.moved) return;

  // a tap: right goes on, left goes back, the middle shows the controls
  const x = e.clientX / W();
  if (S.zoom > 1) {
    if (performance.now() - lastTap < 280) {
      lastTap = 0;
      toggleZoom(e.clientX, e.clientY);
    } else lastTap = performance.now();
    return;
  }
  if (x < 0.3) return turn(-1);
  if (x > 0.7) return turn(1);
  // only original pages can zoom, so only they have to wait for a second tap
  if (isFlow()) return send({ type: "tap" });
  const now = performance.now();
  if (now - lastTap < 280) {
    lastTap = 0;
    clearTimeout(tapTimer);
    toggleZoom(e.clientX, e.clientY);
    return;
  }
  lastTap = now;
  clearTimeout(tapTimer);
  tapTimer = setTimeout(() => send({ type: "tap" }), 240);
});

addEventListener("pointercancel", () => {
  if (P) clearTimeout(P.hold);
  P = null;
});

// a rotated phone: everything's laid out again
let resizeT = 0;
addEventListener("resize", () => {
  clearTimeout(resizeT);
  resizeT = setTimeout(async () => {
    if (!S.kind) return;
    cache.clear();
    S.scrollY = 0;
    if (isFlow()) await relayout();
    else {
      await draw(top, S.page);
      send({ type: "page", page: S.page, total: S.total });
    }
  }, 150);
});
